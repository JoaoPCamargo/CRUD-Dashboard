CREATE TABLE Publisher (
    publisher_id   INTEGER PRIMARY KEY AUTOINCREMENT,
    name           TEXT NOT NULL UNIQUE
);
CREATE TABLE sqlite_sequence(name,seq);
CREATE TABLE Series (
    series_id      INTEGER PRIMARY KEY AUTOINCREMENT,
    name           TEXT NOT NULL,
    publisher_id   INTEGER NOT NULL,
    start_year     INTEGER,
    end_year       INTEGER, total_issues INTEGER,
    FOREIGN KEY (publisher_id) REFERENCES Publisher(publisher_id)
);
CREATE TABLE Arc (
    arc_id         INTEGER PRIMARY KEY AUTOINCREMENT,
    name           TEXT NOT NULL,
    series_id      INTEGER NOT NULL,
    start_issue    INTEGER,
    end_issue      INTEGER,
    FOREIGN KEY (series_id) REFERENCES Series(series_id)
);
CREATE TABLE Comic (
    comic_id       INTEGER PRIMARY KEY AUTOINCREMENT,
    issue_number   INTEGER NOT NULL,
    title          TEXT,
    reading_date   DATE,
    series_id      INTEGER NOT NULL,
    arc_id         INTEGER, rating INTEGER
    CHECK (rating IS NULL OR (rating >= 1 AND rating <= 5)),
    FOREIGN KEY (series_id) REFERENCES Series(series_id),
    FOREIGN KEY (arc_id) REFERENCES Arc(arc_id)
);
CREATE TABLE MangaSeries (
    manga_series_id   INTEGER PRIMARY KEY AUTOINCREMENT,
    name              TEXT NOT NULL UNIQUE,
    author            TEXT NOT NULL
, total_volumes INTEGER);
CREATE TABLE Manga (
    manga_id          INTEGER PRIMARY KEY AUTOINCREMENT,
    volume            INTEGER NOT NULL,
    reading_date      DATE,
    rating            INTEGER CHECK (rating IS NULL OR (rating >= 1 AND rating <= 5)),
    manga_series_id   INTEGER NOT NULL,
    FOREIGN KEY (manga_series_id) REFERENCES MangaSeries(manga_series_id)
);
